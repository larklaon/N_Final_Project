const now = new Date(2026, 9, 18, 12).getTime();
function setup() {
  let n = 0;
  const repo = new LearningRepository(database(), () => `review-${++n}`, () => now);
  repo.installSubject("info-basic");
  repo.updateSettings({ onboarded: true, aiConsentAt: 1 });
  const refresh = jest.fn();
  const publish = () => jest.mocked(useApp).mockReturnValue({
    repo, snapshot: repo.snapshot(), now, refresh, auth: { status: "disconnected" },
    refreshAuth: async () => {}, profiles: [], gapError: null,
  });
  publish();
  return { repo, publish };
}
function answerVisible(ui: ReturnType<typeof render>, repo: LearningRepository) {
  const row = (repo as any).db.getFirstSync("SELECT data FROM learning_sessions LIMIT 1");
  const session = JSON.parse(row.data);
  fireEvent.press(ui.getByText(session.options[0].text));
  fireEvent.press(ui.getByText("확인"));
}
test("after answering, changing subject should immediately offer that subject's question", async () => {
  const { repo, publish } = setup();
  const ui = render(<QuestionScreen />);
  answerVisible(ui, repo);
  repo.installSubject("sqld-taste"); publish();
  ui.rerender(<QuestionScreen />);
  expect(ui.queryByText("지금은 쉬어 가요.")).toBeNull();
  expect(ui.getByText("확인")).toBeTruthy();
});
test("after answering, reset should discard the displayed graded answer", () => {
  const { repo, publish } = setup();
  const ui = render(<QuestionScreen />);
  answerVisible(ui, repo);
  repo.resetCurrent(); publish(); ui.rerender(<QuestionScreen />);
  expect(ui.queryByText("다음")).toBeNull();
  expect(ui.getByText("확인")).toBeTruthy();
});
test("a plan completing after reset must not recreate the erased plan", async () => {
  const { repo } = setup();
  let listener: (event: AiEvent) => void = () => {};
  const start = jest.fn(async () => ({ requestId: "review-plan" }));
  jest.mocked(getSiwc).mockReturnValue({
    startResponse: start, cancelResponse: jest.fn(),
    addListener: (_: string, cb: typeof listener) => { listener = cb; return { remove: jest.fn() }; },
  } as any);
  const pending = requestPlan(repo, { status: "ready", profileId: "p" }, [{ slug: "m", displayName: "M" }], now);
  await waitFor(() => expect(start).toHaveBeenCalledTimes(1));
  repo.resetCurrent();
  await act(async () => listener({ requestId: "review-plan", type: "completed", text: JSON.stringify({
    summary: "old learning history", days: [{ focus: [], newCount: 5, reviewCount: 0, note: "old plan" }],
  }) }));
  await pending;
  expect(repo.plan()).toBeNull();
});
